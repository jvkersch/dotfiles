** use-package

   [[https://github.com/jwiegley/use-package][use-package]] is a package to tidy up your .emacs configuration. It allows you
   to run code before or after a package is loaded, set up key bindings for
   modes, in a neat and concise declarative manner.

 #+BEGIN_SRC emacs-lisp
   (eval-when-compile
      (require 'use-package))
 #+END_SRC

*** Emacs customizations

Turn off all the fluff: no splash screen, short y/n answer format instead of
the default yes/no, no tool bar, etc.

#+BEGIN_SRC emacs-lisp
(setq frame-title-format "Emacs: %f")
(tool-bar-mode -1)
(scroll-bar-mode -1)
(setq inhibit-splash-screen t)
(fset 'yes-or-no-p 'y-or-n-p)
#+END_SRC

Disable C-x C-c to quit emacs, since it is too easy to hit those keys by
mistake. To quit Emacs, use an explicit M-x ~kill-emacs~.

#+BEGIN_SRC emacs-lisp
  (defun disable-exit-keys ()
  (interactive)
    (message "To exit emacs, use M-x kill-emacs"))
  (global-set-key (kbd "C-x C-c") 'disable-exit-keys)
#+END_SRC

*** ace-window

Keybindings to easily and intuitively switch windows.

| Key | Action                                       |
|-----+----------------------------------------------|
| x   | Delete window                                |
| m   | Swap windows                                 |
| M   | Move window                                  |
| c   | Copy window                                  |
| j   | Select buffer                                |
| n   | Select the previous window                   |
| u   | Select buffer in the other window            |
| c   | Split window fairly, vertically/horizontally |
| v   | Split window vertically                      |
| b   | Split window horizontally                    |
| o   | Maximize current window                      |
| ?   | Show these command bindings                  |

#+BEGIN_SRC emacs-lisp
  (use-package ace-window
    :ensure t
    :bind (("M-o" . ace-window)))
#+END_SRC

*** ace-isearch

| M-o <num>         | Jump to window <num>            |
| C-u M-o <num>     | Switch current window and <num> |

** SVG tags

#+BEGIN_SRC emacs-lisp

      (defun jvk/svg-make-colored-tag (tag &optional &rest args)
        (let* ((reverse (or (plist-get args :reverse) nil))
               (foreground (or (plist-get args :foreground)
                               (svg-tag--face-attribute 'svg-tag-default-face :foreground)))
               (background (or (plist-get args :background)
                               (svg-tag--face-attribute 'svg-tag-default-face :background))))
          (if reverse
              (svg-lib-tag tag nil
                     :stroke 0
                     :font-weight 'semibold
                     :foreground background
                     :background foreground)
            (svg-lib-tag tag nil
                   :stroke 2
                   :font-weight 'regular
                   :foreground foreground
                   :background background))))

    (use-package svg-tag-mode
      :config
      (setq svg-tag-tags
        '(("DONE" . ((lambda (tag) (svg-tag-make "DONE" :reverse t))))
          ("TODO" . ((lambda (tag) (jvk/svg-make-colored-tag "TODO"
                                                             :foreground "DodgerBlue1" :reverse t))))
          ("NEXT" . ((lambda (tag) (jvk/svg-make-colored-tag "NEXT" :foreground "orange1" :reverse t))))
          ("WAITING" . ((lambda (tag) (jvk/svg-make-colored-tag "WAIT" :foreground "OrangeRed" :reverse t))))
          ("IN-PROGRESS" . ((lambda (tag) (jvk/svg-make-colored-tag "PROG" :foreground "yellow" :background "black" :reverse t)))))))

  ;; workaround to make SVG tags show in the org agenda
  ;; https://github.com/rougier/svg-tag-mode?tab=readme-ov-file#svg-tags-cannot-render-in-org-agenda
    
    (defun org-agenda-show-svg ()
      (interactive)
      (let* ((case-fold-search nil)
             (keywords (mapcar #'svg-tag--build-keywords svg-tag--active-tags))
             (keyword (car keywords)))
        (while keyword
          (save-excursion
            (while (re-search-forward (nth 0 keyword) nil t)
              (overlay-put (make-overlay
                            (match-beginning 0) (match-end 0))
                           'display  (nth 3 (eval (nth 2 keyword)))) ))
          (pop keywords)
          (setq keyword (car keywords)))))
    (add-hook 'org-agenda-finalize-hook #'org-agenda-show-svg)
